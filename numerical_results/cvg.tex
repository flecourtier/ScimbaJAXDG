% Tracer une série avec style méthode+degré du registre
% Utilise \cvg@currentcol au moment de l'appel
\newcommand{\cvg@plotone}[4]{%
    \edef\cvg@col{\cvg@currentcol}%
    \addplot[/cvg/method/#1, /cvg/degree/#2, mark options={solid}]
    table[x=\cvg@xcol, y=\cvg@col, col sep=comma]{#3};%
    \ifcvg@showlegend
        \addlegendentry{\,#4\;}%
    \fi
}

% Tracer une série avec style libre
\newcommand{\cvg@plotraw}[3]{%
    \edef\cvg@col{\cvg@currentcol}%
    \addplot[#3, mark options={solid}]
    table[x=\cvg@xcol, y=\cvg@col, col sep=comma]{#1};%
    \ifcvg@showlegend
        \addlegendentry{\,#2\;}%
    \fi
}

% Fixer la colonne CSV et le label selon la norme
\newcommand{\cvg@setnorm}[1]{%
    \ifstrequal{#1}{L2}{%
        \gdef\cvg@currentcol{errL2}%
        \gdef\cvg@currentlabel{$L^2$}%
    }{}%
    \ifstrequal{#1}{H1semi}{%
        \gdef\cvg@currentcol{errH1semi}%
        \gdef\cvg@currentlabel{Semi-$H^1$}%
    }{}%
    \ifstrequal{#1}{Linf}{%
        \gdef\cvg@currentcol{errLinf}%
        \gdef\cvg@currentlabel{$L^\infty$}%
    }{}%
    \ifstrequal{#1}{Cond}{%
        \gdef\cvg@currentcol{cond}%
        \gdef\cvg@currentlabel{$\kappa$}%
    }{}%
}

% \addseries[label=..., type=Q]{méthode}{degré}{fichier.csv}
%   label : remplace le NOM de la méthode (le P/Q_deg est toujours ajouté)
%   type  : P (défaut) ou Q
\newcommand{\addseries}[4][]{%
    \def\cvg@label{}\def\cvg@type{P}%
    \pgfkeys{/cvg/series, #1}%
    % Nom affiché : label= s'il est fourni, sinon le nom de méthode
    \ifx\cvg@label\empty \def\cvg@nameonly{#2}\else \let\cvg@nameonly\cvg@label\fi%
    % Label complet : nom + P/Q_deg (\unexpanded protège le math dans \protected@edef)
    \def\cvg@Qtmp{Q}%
    \ifx\cvg@type\cvg@Qtmp
        \protected@edef\cvg@label{\cvg@nameonly\unexpanded{ $\mathbb{Q}_{#3}$}}%
    \else
        \protected@edef\cvg@label{\cvg@nameonly\unexpanded{ $\mathbb{P}_{#3}$}}%
    \fi
    % Chemin complet : préfixe si \setcsvdir a été appelé
    \ifx\cvg@csvdir\empty
        \edef\cvg@fullpath{#4}%
    \else
        \edef\cvg@fullpath{\cvg@csvdir/#4}%
    \fi
    \ifx\cvg@reffile\empty \xdef\cvg@reffile{\cvg@fullpath}\fi%
    \cvg@toks=\expandafter{\cvg@label}%
    \protected@edef\cvg@tmp{%
        \unexpanded\expandafter{\the\cvg@serieslist}%
        \noexpand\cvg@plotone{#2}{#3}{\cvg@fullpath}{\the\cvg@toks}%
    }%
    \global\cvg@serieslist=\expandafter{\cvg@tmp}%
}

% \addrawseries{fichier.csv}{label légende}{options pgfplots}
%   Style entièrement libre (hors registre).
\newcommand{\addrawseries}[3]{%
    \ifx\cvg@csvdir\empty
        \edef\cvg@fullpath{#1}%
    \else
        \edef\cvg@fullpath{\cvg@csvdir/#1}%
    \fi
    \ifx\cvg@reffile\empty \xdef\cvg@reffile{\cvg@fullpath}\fi%
    \protected@edef\cvg@tmp{%
        \unexpanded\expandafter{\the\cvg@serieslist}%
        \noexpand\cvg@plotraw{\cvg@fullpath}{#2}{#3}%
    }%
    \global\cvg@serieslist=\expandafter{\cvg@tmp}%
}

%
% Crée un graphique de convergence autonome (tikzpicture).
% À placer dans une subfigure ou seul.
%
% Arguments :
%   [legendcols] : colonnes de légende (défaut : 1)
%   {norme}      : L2, H1semi, Linf ou Cond
%
% Contenu :
%   \addseries[label=..., type=Q]{méthode}{degré}{fichier.csv}
%     label auto si omis : "méthode $\mathbb{P/Q}_{degré}$"
%     type=P (défaut) ou Q

%   \setcsvdir{dossier}             (optionnel, préfixe pour tous les chemins CSV)
%   \setslopes{pentes}{ypos}        (fonctions croissantes, triangle à droite)
%   \setslopesdec{pentes}{ypos}     (fonctions décroissantes, triangle à gauche)
%   \setxtickrotate[yshift]{angle}  (optionnel, ex: \setxtickrotate[-5pt]{45})
%   \setxcol{colonne}{label}        (optionnel, ex: \setxcol{n_cells}{$n_\text{cells}$})
%   \setxlimits{min}{max}           (optionnel, ex: \setxlimits{1e-3}{1e-1})
%   \setylimits{min}{max}           (optionnel, ex: \setylimits{1e-6}{1e-1})
%
% Exemple :
%   \begin{cvgh}[2]{L2}
%     \addseries{DG SIPG}{1}{results/DG_SIPG_P1.csv}
%     \addseries[type=Q]{DG NIPG}{1}{results/DG_NIPG_P1.csv}
%     \addseries[label=BZ custom]{DG BZ}{1}{results/DG_BZ_P1.csv}  % → "BZ custom $\mathbb{P}_1$"
%     \setslopes{2}{1e-4}

\newenvironment{cvgh}[2][1]{%
% Incrémenter le compteur global
\pgfmathtruncatemacro{\cvg@iter}{\cvg@iter+1}%
\xdef\cvg@iter{\cvg@iter}%
% Réinitialiser l'état
\global\cvg@serieslist={}%
\gdef\cvg@reffile{}%
\gdef\cvg@slopes{}%
\gdef\cvg@ypos{}%
\gdef\cvg@slopesdec{}%
\gdef\cvg@yposdec{}%
\gdef\cvg@xticklabrot{}%
\gdef\cvg@xticklabshift{0pt}%
\gdef\cvg@xmin{}%
\gdef\cvg@xmax{}%
\gdef\cvg@ymin{}%
\gdef\cvg@ymax{}%
\gdef\cvg@xcol{h}%
\gdef\cvg@xlabel{$h$}%
\gdef\cvg@csvdir{}%
\global\cvg@showlegendtrue%
\global\cvg@placelegendtrue%
\gdef\cvg@legendcols{#1}%
% Fixer la norme
\cvg@setnorm{#2}%
}{%
% Lire le CSV de référence pour les ticks en x et les pentes
\pgfplotstableread[col sep=comma]{\cvg@reffile}\cvg@datatable%
\pgfplotstablegetelem{0}{\cvg@xcol}\of\cvg@datatable%
\pgfmathsetmacro{\cvg@hfirst}{\pgfplotsretval}%
\pgfplotstablegetelem{1}{\cvg@xcol}\of\cvg@datatable%
\pgfmathsetmacro{\cvg@hsecond}{\pgfplotsretval}%
\pgfmathsetmacro{\cvg@hdiff}{\cvg@hfirst - \cvg@hsecond}%
\ifx\cvg@slopesdec\empty\else
    \pgfplotstablegetrowsof{\cvg@datatable}%
    \pgfmathtruncatemacro{\cvg@lastidx}{\pgfplotsretval-1}%
    \pgfmathtruncatemacro{\cvg@prevlastidx}{\pgfplotsretval-2}%
    \pgfplotstablegetelem{\cvg@lastidx}{\cvg@xcol}\of\cvg@datatable%
    \pgfmathsetmacro{\cvg@hlast}{\pgfplotsretval}%
    \pgfplotstablegetelem{\cvg@prevlastidx}{\cvg@xcol}\of\cvg@datatable%
    \pgfmathsetmacro{\cvg@hprevlast}{\pgfplotsretval}%
    \pgfmathsetmacro{\cvg@hdifffromleft}{\cvg@hprevlast - \cvg@hlast}%
\fi
%
\resizebox{\linewidth}{!}{%
    \begin{tikzpicture}[every node/.style={inner sep=0.5, outer sep=0.5}]
        \ifx\cvg@xticklabrot\empty\else
            \pgfplotsset{xticklabel style={rotate=\cvg@xticklabrot, anchor=east, yshift=\cvg@xticklabshift}}%
        \fi
        \ifx\cvg@xmin\empty\else
            \pgfplotsset{xmin=\cvg@xmin, xmax=\cvg@xmax}%
        \fi
        \ifx\cvg@ymin\empty\else
            \pgfplotsset{ymin=\cvg@ymin, ymax=\cvg@ymax}%
        \fi
        \begin{loglogaxis}[
                smaller labels,
                name=cvgaxis\cvg@iter,
                axis lines=left,
                enlarge x limits={abs=10pt},
                enlarge y limits={abs=10pt},
                xlabel={\cvg@xlabel},
                ylabel={\rotatebox{270}{\cvg@currentlabel}},
                xlabel style={at={(ticklabel* cs:1.01)}, anchor=south west},
                ylabel style={at={(ticklabel* cs:1.01)}, anchor=west},
                xtick=data,
                xticklabels from table={\cvg@datatable}{\cvg@xcol},
                width=\cvgplotwidth, height=\cvgplotheight,
                mark options={solid, scale=1},
                ytick style={draw=none},
                grid=major,
                legend columns=\cvg@legendcols,
                legend to name=cvgleg\cvg@iter,
                legend image post style={mark options={solid, scale=1}},
            ]
            % Triangles de pente (si définis)
            \ifx\cvg@slopes\empty\else
                \expandafter\printslopeinv\expandafter{\cvg@slopes}%
                {\cvg@hsecond}{\cvg@ypos}{\cvg@hdiff}%
            \fi
            \ifx\cvg@slopesdec\empty\else
                \expandafter\printslope\expandafter{\cvg@slopesdec}%
                {\cvg@hlast}{\cvg@yposdec}{\cvg@hdifffromleft}%
            \fi
            % Tracer toutes les séries
            \the\cvg@serieslist
        \end{loglogaxis}
        \ifcvg@showlegend
            \ifcvg@placelegend
                \node[yshift=-20pt] at (cvgaxis\cvg@iter.outer south)
                {\pgfplotslegendfromname{cvgleg\cvg@iter}};
            \fi
        \fi
    \end{tikzpicture}%
}%
}
